export function parseBotId(id: string): { ticker: string; trigger: string } | null {
  const m = /^([A-Z]+)-([A-Z0-9]+)$/.exec(id)
  return m ? { ticker: m[1], trigger: m[2] } : null
}

export function towerPosition(ti: number, ki: number): [number, number, number] {
  const a = (ti * 2 * Math.PI) / 3 + Math.PI / 2
  const R = 7
  const off = (ki - 1) * 2.4
  return [Math.cos(a) * R - Math.sin(a) * off, 0, Math.sin(a) * R + Math.cos(a) * off]
}

export const SLOW_SEC = 90
export const STALE_SEC = 180

export function formatPnl(n: number): string {
  const abs = Math.abs(n).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
  return n > 0 ? `+$${abs}` : n < 0 ? `-$${abs}` : `$${abs}`
}
export const formatPct = (n: number | null | undefined) => (n == null ? '-' : `${Math.round(n)}%`)
export const formatNum = (n: number | undefined | null, d = 2) => (typeof n === 'number' ? n.toFixed(d) : '-')
export const formatTime = (iso: string | null) => (iso && iso.length >= 19 ? iso.slice(11, 19) : '-')

function etParts(now: Date) {
  const p = new Intl.DateTimeFormat('en-US', {
    timeZone: 'America/New_York', weekday: 'short', hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: false,
  }).formatToParts(now)
  const get = (t: string) => p.find(x => x.type === t)!.value
  return { wd: get('weekday'), sec: (+get('hour') % 24) * 3600 + +get('minute') * 60 + +get('second') }
}

export type Phase = { phase: 'pre' | 'open' | 'closed'; label: string; seconds: number }
export function marketPhase(now: Date): Phase {
  const { wd, sec } = etParts(now)
  const OPEN = 9.5 * 3600, CLOSE = 16 * 3600
  if (wd === 'Sat' || wd === 'Sun') return { phase: 'closed', label: 'Closed (weekend)', seconds: 0 }
  if (sec < OPEN) return { phase: 'pre', label: 'Opens 9:30 ET', seconds: OPEN - sec }
  if (sec < CLOSE) return { phase: 'open', label: 'Market open', seconds: CLOSE - sec }
  return { phase: 'closed', label: 'Market closed', seconds: 0 }
}

export type Fresh = 'live' | 'slow' | 'stale' | 'closed' | 'none'
export function freshness(ageSec: number | null, p: Pick<Phase, 'phase'>): Fresh {
  if (ageSec == null) return 'none'
  if (p.phase !== 'open') return ageSec > STALE_SEC ? 'closed' : 'live'
  return ageSec > STALE_SEC ? 'stale' : ageSec > SLOW_SEC ? 'slow' : 'live'
}

import type { Inputs, Side } from './types'

export function inferSide(bot: string, inputs: Inputs): Side | null {
  const trigger = bot.split('-')[1]
  const level = trigger === 'VWAP' ? inputs.vwap : trigger === 'EMA50' ? inputs.ema50 : undefined
  const price = inputs.price
  if (typeof level !== 'number' || typeof price !== 'number' || price === level) return null
  return price < level ? 'CALL' : 'PUT'
}
export function parseSignalSide(text: string): Side | null {
  const m = /\b(CALL|PUT)\b/.exec(text ?? '')
  return m ? (m[1] as Side) : null
}

export const stateLabel = (s: string) => STATE_LABEL[s] ?? s
export const stateIcon = (s: string) => STATE_ICON[s] ?? '?'

export const beamFraction = (c: number | null): number =>
  c == null || Number.isNaN(c) ? 0 : Math.min(1, Math.max(0, c / 100))

export type Effect = 'none' | 'ring' | 'ring-static' | 'pulse' | 'flicker'
export interface TowerVisual { color: string; glow: number; beam: number; effect: Effect; hatched: boolean; label: string | null }

export function visualFor(row: BotRow, reduced: boolean): TowerVisual {
  const sc = sideColor(row.side)

```ts
import type { AppState, BotEvent, BotRow, Mark, ServerMsg, SideSource, Side, SnapshotBot } from './types'
import { inferSide, parseSignalSide } from './side'

export const FEED_MAX = 200
export const TICKS_MAX = 390
export const MARKS_MAX = 100

export const initialState: AppState = { bots: {}, lastSeq: 0, feed: [], ticks: {}, marks: {}, lastEventAt: null }

interface SideState { side: Side | null; sideSource: SideSource }
const NONE: SideState = { side: null, sideSource: 'none' }
const CLEARING = new Set(['idle', 'cooldown', 'locked', 'paused', 'stale'])

function applySide(prev: SideState, ev: BotEvent): SideState {
  if (ev.side === 'CALL' || ev.side === 'PUT') return { side: ev.side, sideSource: 'explicit' }
  if (ev.side === null) return NONE
  if (ev.reason_code === 'SIGNAL') {
    const s = parseSignalSide(ev.reason_text)
    if (s) return { side: s, sideSource: 'signal' }
  }
  if (CLEARING.has(ev.state)) return NONE
  if (ev.state === 'forming') {
    const s = inferSide(ev.bot, ev.inputs ?? {})
    return s ? { side: s, sideSource: 'inferred' } : NONE
  }
  return prev
}

function rowFromSnapshot(b: SnapshotBot): BotRow {
  const explicit = b.side === 'CALL' || b.side === 'PUT'
  let side: SideState = explicit ? { side: b.side as Side, sideSource: 'explicit' } : NONE
  if (!explicit && b.state === 'forming') {
    const s = inferSide(b.bot, b.inputs ?? {})
    if (s) side = { side: s, sideSource: 'inferred' }
  }
  return {
    bot: b.bot, ticker: b.ticker ?? b.bot.split('-')[0], state: b.state ?? 'idle', reason_code: b.reason_code ?? '',
    reason_text: b.reason_text ?? '', charge_pct: b.charge_pct ?? null, inputs: b.inputs ?? {}, time: b.time ?? null,
    pnl_day: typeof b.pnl_day === 'number' ? b.pnl_day : 0, trades: typeof b.trades === 'number' ? b.trades : 0,
    paused: !!b.paused, ...side,
  }
}

function addMark(marks: Record<string, Mark[]>, ev: BotEvent): Record<string, Mark[]> {
  let type: Mark['type'] | null = null
  if (ev.kind === 'fill' || ev.kind === 'order_sent') {
    if (ev.reason_code === 'ENTRY_FILL') type = 'entry'
    else if (ev.reason_code === 'EXIT_FILL' || ev.reason_code === 'PARTIAL_FILL') type = 'exit'
  } else if (ev.kind === 'trade_closed') type = 'exit'
  const price = ev.fill_price ?? ev.inputs?.price
  if (!type || typeof price !== 'number') return marks
  const list = [...(marks[ev.ticker] ?? []), { time: ev.time, price, type, bot: ev.bot }].slice(-MARKS_MAX)
  return { ...marks, [ev.ticker]: list }
}

export function reduce(state: AppState, msg: ServerMsg, nowMs: number): AppState {
  if (msg.type === 'snapshot') {
    const bots: Record<string, BotRow> = {}
    for (const [id, b] of Object.entries(msg.bots ?? {})) bots[id] = rowFromSnapshot({ ...b, bot: b.bot ?? id })
    return { ...state, bots, lastSeq: typeof msg.last_seq === 'number' ? msg.last_seq : 0, lastEventAt: nowMs }
  }
  const ev = msg.event
  if (!ev || typeof ev.seq !== 'number' || ev.seq <= state.lastSeq) return state
  const row = state.bots[ev.bot]
  const base: AppState = { ...state, lastSeq: ev.seq, lastEventAt: nowMs }
  if (!row) return base

  const r: BotRow = { ...row }
  if (typeof ev.state === 'string' && ev.state !== '') {
    r.state = ev.state
    r.reason_code = ev.reason_code ?? ''
    r.reason_text = ev.reason_text ?? ''
    r.charge_pct = ev.charge_pct ?? null
    r.time = ev.time ?? r.time
    if (ev.inputs && Object.keys(ev.inputs).length) r.inputs = ev.inputs
    Object.assign(r, applySide(row, ev))
  }
  if (typeof ev.pnl_day === 'number') r.pnl_day = ev.pnl_day
  if (typeof ev.trades === 'number') r.trades = ev.trades
  if (typeof ev.paused === 'boolean') r.paused = ev.paused

  const next: AppState = { ...base, bots: { ...state.bots, [ev.bot]: r } }
  if (ev.kind !== 'tick') next.feed = [ev, ...state.feed].slice(0, FEED_MAX)
  if (ev.kind === 'tick' && typeof ev.inputs?.price === 'number') {
    const pts = state.ticks[ev.ticker] ?? []
    if (!pts.length || pts[pts.length - 1].time !== ev.time)
      next.ticks = { ...state.ticks, [ev.ticker]: [...pts, { time: ev.time, price: ev.inputs.price }].slice(-TICKS_MAX) }
  }
  next.marks = addMark(state.marks, ev)
  return next
}

export const totalPnl = (s: AppState) => Object.values(s.bots).reduce((a, b) => a + b.pnl_day, 0)
export const leaderboard = (s: AppState): BotRow[] =>
  Object.values(s.bots).sort((a, b) => b.pnl_day - a.pnl_day || a.bot.localeCompare(b.bot))
```

function startTrade(b) {

function finishTrade(b) {
  const o = b.pos.outcome
  const final = o === 'win' ? 180 + rand() * 300 : o === 'loss' ? -(60 + rand() * 60) : 20 + rand() * 40
  const code = o === 'win' ? (rand() > 0.5 ? 'EXIT_TARGET' : 'EXIT_LOCK') : o === 'loss' ? 'EXIT_STOP' : 'EXIT_TIME'
  b.pos.pnl = +final.toFixed(2)
  set(b, 'exiting', code, `${code.toLowerCase().replace('_', ' ')}, ${b.pos.pnl}`, null)
  emit(b, 'state_change')
  emit(b, 'order_sent', { qty: 10 })
  b.reason_code = 'EXIT_FILL'; b.reason_text = 'exit filled'
  emit(b, 'fill', { fill_price: +TK[b.ticker].p.toFixed(2), fill_source: 'modeled', qty: 10 })
  b.realised = +(b.realised + b.pos.pnl).toFixed(2); b.trades++
  const pnl = b.pos.pnl; b.pos = null
  b.reason_code = code
  emit(b, 'trade_closed', { pnl })

}

function stepBot(b) {
  if (b.paused || b.stale) return emit(b, 'tick')
  b.t++
  switch (b.phase) {
    case 'pre':
      set(b, 'idle', 'OUTSIDE_WINDOW', 'before trading window')
      if (b.t >= 4) { b.phase = 'rangeform'; b.t = 0 }
      break
    case 'rangeform':
      set(b, 'idle', 'RANGE_FORMING', 'opening range still forming')
      if (b.t >= 4) { b.phase = 'watch'; b.t = 0; b.watchLen = 3 }
      break
    case 'watch':
      if (b.trigger === 'VWAP' && b.t === 2) set(b, 'idle', 'CHOP_SQUEEZE', 'EMA9/EMA21 squeezed, skipping VWAP cross')
      else set(b, 'idle', 'WATCHING', 'watching for a signal', b.trigger === 'RANGE' ? null : Math.round(5 + rand() * 10))
      if (b.t >= b.watchLen) { b.phase = 'forming'; b.t = 0; b.side = rand() > 0.5 ? 'CALL' : 'PUT' }
      break
    case 'forming': {
      const c = Math.min(100, 15 + b.t * 11)
      set(b, 'forming', 'WATCHING', `${c}% charged`, c)
      if (b.t >= 8) { b.charge = 100; return startTrade(b), undefined }
      break
    }
    case 'managing': {
      const o = b.pos.outcome, k = b.t / b.pos.len
      b.pos.pnl = +((o === 'loss' ? -90 : o === 'win' ? 250 : 40) * k + gauss() * 15).toFixed(2)
      set(b, 'managing', 'HOLDING', `holding ${b.side}, open P&L ${b.pos.pnl}`)
      if (b.t >= b.pos.len) return finishTrade(b), undefined
      break
    }
    case 'cooldown':
      set(b, 'cooldown', 'COOLDOWN', 'cooling down after trade')
      if (b.t >= 4) { b.phase = 'watch'; b.t = 0; b.watchLen = 5 + (b.trades % 4); set(b, 'idle', 'WATCHING', 'watching for a signal'); emit(b, 'state_change'); return }
      break

      if (--b.lockedFor <= 0) { b.phase = 'watch'; b.t = 0; b.watchLen = 4; b.realised = 0; b.trades = 0 }
      break
  }
  const before = b.state
  emit(b, 'tick')
  return before
}

import type { ServerMsg } from '../model/types'

export type Status = 'connecting' | 'live' | 'down'
export function backoffDelay(attempt: number, rand: () => number = Math.random): number {
  const base = Math.min(15000, 1000 * 2 ** attempt)
  return Math.max(500, Math.round(base * (0.5 + rand() * 0.5)))
}
const base = (import.meta as any).env?.VITE_API_BASE as string | undefined
export const apiUrl = (p: string) => (base ? base.replace(/\/$/, '') + p : p)
const wsUrl = () => {
  if (base) return base.replace(/^http/, 'ws').replace(/\/$/, '') + '/ws'
  return `${location.protocol === 'https:' ? 'wss' : 'ws'}://${location.host}/ws`
}

export function connectStream(h: { onMessage: (m: ServerMsg) => void; onStatus: (s: Status) => void }): () => void {
  let ws: WebSocket | null = null, attempt = 0, timer: ReturnType<typeof setTimeout> | undefined, closed = false
  const open = () => {
    if (closed) return
    h.onStatus('connecting')
    ws = new WebSocket(wsUrl())
    ws.onopen = () => { attempt = 0; h.onStatus('live') }
    ws.onmessage = e => {
      try { const m = JSON.parse(e.data); if (m && (m.type === 'snapshot' || m.type === 'event')) h.onMessage(m) } catch { /* ignore bad frame */ }
    }
    ws.onclose = () => { if (closed) return; h.onStatus('down'); timer = setTimeout(open, backoffDelay(attempt++)) }
    ws.onerror = () => ws?.close()
  }
  open()
  return () => { closed = true; clearTimeout(timer); ws?.close() }
}

import { useEffect, useReducer, useState } from 'react'
import { initialState, reduce } from '../model/reducer'
import type { AppState, ServerMsg } from '../model/types'
import { connectStream, type Status } from './socket'

export function useStream(): { state: AppState; status: Status } {
  const [state, dispatch] = useReducer((s: AppState, a: { msg: ServerMsg; now: number }) => reduce(s, a.msg, a.now), initialState)
  const [status, setStatus] = useState<Status>('connecting')
  useEffect(() => connectStream({ onMessage: msg => dispatch({ msg, now: Date.now() }), onStatus: setStatus }), [])
  return { state, status }
}
